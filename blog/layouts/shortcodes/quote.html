Markdown is supported. */ -}} {{- $name := .Get "name" -}} {{- $title := .Get "title" -}} {{- $company := .Get "company" -}} {{- $logo := .Get "logo" -}} {{- $logoDark := .Get "logo-dark" -}} {{- $variants := slice -}} {{- with $logo -}} {{- if $logoDark -}} {{- $variants = slice (dict "file" .
